#include <WiFi.h>
#include <WebServer.h>
#include <ESP32Servo.h>

// Hardware pins
#define IR_ENTRY  18
#define PIR_EXIT  19
#define SERVO_PIN 25
#define GREEN_LED 26
#define RED_LED   27

Servo barrierServo;

const char* ssid     = "Eliyas";
const char* password = "0123456789";

WebServer server(80);

// --- State ---
int totalSlots     = 4;
int availableSlots = 4;
int flagEntry      = 0;
int flagExit       = 0;
String gateStatus  = "Closed";

// Admin controls
bool systemLocked     = false;  // locks all sensor detection
bool gateOverride     = false;  // admin manually opened gate
String lastActivity   = "System started";

// PIR timing
unsigned long pirHighStart      = 0;
bool pirTriggered               = false;
const unsigned long PIR_HOLD_MS = 300;

// Entry cooldown
unsigned long lastEntryTime          = 0;
const unsigned long EXIT_COOLDOWN_MS = 5000;

// ---------------- Helpers ----------------
void openGate() {
  gateStatus = "Open";
  barrierServo.write(0);
}

void closeGate() {
  barrierServo.write(90);
  gateStatus = "Closed";
}

void updateLEDs() {
  if (availableSlots > 0) {
    digitalWrite(GREEN_LED, HIGH);
    digitalWrite(RED_LED,   LOW);
  } else {
    digitalWrite(GREEN_LED, LOW);
    digitalWrite(RED_LED,   HIGH);
  }
}

// ---------------- API Handlers ----------------
void handleData() {
  String json = "{";
  json += "\"available\":"   + String(availableSlots) + ",";
  json += "\"occupied\":"    + String(totalSlots - availableSlots) + ",";
  json += "\"total\":"       + String(totalSlots) + ",";
  json += "\"gate\":\""      + gateStatus + "\",";
  json += "\"locked\":"      + String(systemLocked ? "true" : "false") + ",";
  json += "\"activity\":\""  + lastActivity + "\",";
  json += "\"occupied_slots\":[";
  int occ = totalSlots - availableSlots;
  for (int i = 1; i <= occ; i++) {
    json += String(i);
    if (i < occ) json += ",";
  }
  json += "]}";
  server.send(200, "application/json", json);
}

// Admin: Open gate manually
void handleAdminGateOpen() {
  if (!gateOverride) {
    gateOverride = true;
    openGate();
    lastActivity = "Admin opened gate";
    Serial.println("[ADMIN] Gate opened manually");
  }
  server.send(200, "application/json", "{\"status\":\"gate_opened\"}");
}

// Admin: Close gate manually
void handleAdminGateClose() {
  if (gateOverride) {
    gateOverride = false;
    closeGate();
    lastActivity = "Admin closed gate";
    Serial.println("[ADMIN] Gate closed manually");
  }
  server.send(200, "application/json", "{\"status\":\"gate_closed\"}");
}

// Admin: Lock system (ignore sensors)
void handleAdminLock() {
  systemLocked = true;
  closeGate();
  lastActivity = "System locked by admin";
  Serial.println("[ADMIN] System locked");
  server.send(200, "application/json", "{\"status\":\"locked\"}");
}

// Admin: Unlock system
void handleAdminUnlock() {
  systemLocked = false;
  lastActivity = "System unlocked by admin";
  Serial.println("[ADMIN] System unlocked");
  server.send(200, "application/json", "{\"status\":\"unlocked\"}");
}

// Admin: Add a slot manually (e.g. override wrong count)
void handleAdminSlotAdd() {
  if (availableSlots < totalSlots) {
    availableSlots++;
    updateLEDs();
    lastActivity = "Admin added 1 slot";
    Serial.println("[ADMIN] Slot added");
  }
  server.send(200, "application/json", "{\"status\":\"slot_added\"}");
}

// Admin: Remove a slot manually
void handleAdminSlotRemove() {
  if (availableSlots > 0) {
    availableSlots--;
    updateLEDs();
    lastActivity = "Admin removed 1 slot";
    Serial.println("[ADMIN] Slot removed");
  }
  server.send(200, "application/json", "{\"status\":\"slot_removed\"}");
}

// Admin: Reset slots to full
void handleAdminReset() {
  availableSlots = totalSlots;
  updateLEDs();
  lastActivity = "Admin reset all slots";
  Serial.println("[ADMIN] Slots reset to full");
  server.send(200, "application/json", "{\"status\":\"reset\"}");
}

// ---------------- Setup ----------------
void setup() {
  Serial.begin(115200);
  pinMode(IR_ENTRY,  INPUT);
  pinMode(PIR_EXIT,  INPUT);
  pinMode(GREEN_LED, OUTPUT);
  pinMode(RED_LED,   OUTPUT);

  barrierServo.attach(SERVO_PIN);
  barrierServo.write(90);
  updateLEDs();

  WiFi.begin(ssid, password);
  Serial.print("Connecting");
  while (WiFi.status() != WL_CONNECTED) { delay(500); Serial.print("."); }
  Serial.println("\nIP: " + WiFi.localIP().toString());

  // Routes
  server.on("/",                  []() { server.send(200, "text/html", getHTML()); });
  server.on("/data",              handleData);
  server.on("/admin/gate/open",   handleAdminGateOpen);
  server.on("/admin/gate/close",  handleAdminGateClose);
  server.on("/admin/lock",        handleAdminLock);
  server.on("/admin/unlock",      handleAdminUnlock);
  server.on("/admin/slot/add",    handleAdminSlotAdd);
  server.on("/admin/slot/remove", handleAdminSlotRemove);
  server.on("/admin/reset",       handleAdminReset);

  server.begin();
  lastActivity = "System online";
}

// ---------------- Loop ----------------
void loop() {
  server.handleClient();

  // Skip sensor logic if locked or gate is in admin override
  if (systemLocked || gateOverride) return;

  // --- Entry: IR sensor ---
  if (digitalRead(IR_ENTRY) == LOW && flagEntry == 0) {
    flagEntry = 1;
    digitalWrite(GREEN_LED, LOW);
    digitalWrite(RED_LED,   HIGH);

    if (availableSlots > 0) {
      openGate();
      delay(2000);
      closeGate();
      availableSlots--;
      lastEntryTime = millis();
      lastActivity  = "Car entered";
      Serial.println("[ENTRY] Car entered. Slots left: " + String(availableSlots));
    } else {
      lastActivity = "Entry blocked - full";
      Serial.println("[ENTRY] Parking full");
    }
    updateLEDs();
  }
  if (digitalRead(IR_ENTRY) == HIGH && flagEntry == 1) {
    flagEntry = 0;
    updateLEDs();
  }

  // --- Exit: PIR sensor with cooldown ---
  int pirState = digitalRead(PIR_EXIT);

  if (pirState == HIGH) {
    if (!pirTriggered) {
      pirHighStart = millis();
      pirTriggered = true;
    } else {
      bool cooldownOver = (millis() - lastEntryTime) >= EXIT_COOLDOWN_MS;
      if ((millis() - pirHighStart) >= PIR_HOLD_MS && flagExit == 0 && cooldownOver) {
        flagExit = 1;
        if (availableSlots < totalSlots) {
          openGate();
          delay(2000);
          closeGate();
          availableSlots++;
          lastActivity = "Car exited";
          Serial.println("[EXIT] Car exited. Slots left: " + String(availableSlots));
        }
        updateLEDs();
      }
    }
  } else {
    pirTriggered = false;
    pirHighStart = 0;
    flagExit     = 0;
  }
}
